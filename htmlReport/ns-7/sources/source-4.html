 @OptIn(ExperimentalMaterial3Api::class)
 @Composable
 fun OrderHistoryScreen(
     orderViewModel: OrderViewModel,
     userId: Long,
     onBack: () -> Unit
 ) {
     val orders by orderViewModel.getOrdersByUser(userId).collectAsState(initial = emptyList())
     
     LaunchedEffect(userId) {
         orderViewModel.refreshOrders()
     }
     
     Scaffold(
         topBar = {
             TopAppBar(
                 title = { Text("Historial de Compras") },
                 navigationIcon = {
                     IconButton(onClick = onBack) {
                         Icon(Icons.Filled.ArrowBack, contentDescription = "Atrás")
                     }
                 }
             )
         }
     ) { padding ->
         if (orders.isEmpty()) {
             Box(
                 modifier = Modifier
                     .fillMaxSize()
                     .padding(padding),
                 contentAlignment = Alignment.Center
             ) {
                 Column(
                     horizontalAlignment = Alignment.CenterHorizontally,
                     verticalArrangement = Arrangement.spacedBy(8.dp)
                 ) {
                     Text(
                         text = "No hay pedidos",
                         style = MaterialTheme.typography.titleLarge,
                         color = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.6f)
                     )
                     Text(
                         text = "Tus compras aparecerán aquí",
                         style = MaterialTheme.typography.bodyMedium,
                         color = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.4f)
                     )
                 }
             }
         } else {
             LazyColumn(
                 modifier = Modifier
                     .fillMaxSize()
                     .padding(padding)
                     .padding(16.dp),
                 verticalArrangement = Arrangement.spacedBy(12.dp)
             ) {
                 items(orders) { orderWithProduct ->
                     OrderCard(orderWithProduct)
                 }
             }
         }
     }
 }

 @Composable
 fun StatusChip(estado: String) {
     val color = when (estado) {
         "confirmado" -> MaterialTheme.colorScheme.primary
         "pendiente" -> MaterialTheme.colorScheme.tertiary
         "entregado" -> MaterialTheme.colorScheme.primaryContainer
         else -> MaterialTheme.colorScheme.error
     }
     
     Surface(
         color = color,
         shape = RoundedCornerShape(16.dp)
     ) {
         Text(
             text = estado.replaceFirstChar { it.uppercase() },
             modifier = Modifier.padding(horizontal = 12.dp, vertical = 4.dp),
             style = MaterialTheme.typography.labelSmall,
             color = MaterialTheme.colorScheme.onPrimary
         )
     }
